                 "ftg" = "TG, mg/dL", "tc" = "TC, mg/dL", "hdl" = "HDL, mg/dL",

## Fasting glycemic measures
nhanes_ndi %>% 
  select(SEQN, fastinglabs) %>% 
  pivot_longer(-SEQN, names_to = "FastingLabs") %>% 
  filter(!is.na(value)) %>% 
  mutate_at("FastingLabs", ~factor(., levels = names(recode_labs))) %>%
  filter(FastingLabs %in% c("fg","fi", "ogtt_2hg_cleaned")) %>% 
  mutate(FastingLabs = recode(FastingLabs, !!!recode_labs)) %>%
  ## Filter to subset of indices
  ggplot(aes(x=value)) + 
  facet_wrap(~FastingLabs, scale = "free") + 
  geom_histogram() + 
  ylab("Distribution, n") + xlab("Fasting blood levels") +
  ggplot_theme_basic
